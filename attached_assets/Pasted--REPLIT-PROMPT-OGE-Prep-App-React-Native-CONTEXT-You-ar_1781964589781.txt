### Question system — CRITICAL LOGIC:

Implement a proper shuffle + anti-repeat system:

```typescript
// Answer options must NEVER repeat in the same session
// Use Fisher-Yates shuffle on options array
// Track shown questions in session state
// Wrong answer → question returns to queue (shuffled position)
// Correct answer → question removed from queue
// Never show the same distractor in the same position twice in a row

interface Question {
  id: string;
  text: string;
  correctAnswer: string;
  distractors: string[]; // min 3, max 5 wrong options
  explanation: string; // shown after answering
  difficulty: 'easy' | 'medium' | 'hard';
  topic: string;
}

function shuffleAnswers(question: Question): string[] {
  const all = [question.correctAnswer, ...question.distractors];
  // Fisher-Yates shuffle
  for (let i = all.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [all[i], all[j]] = [all[j], all[i]];
  }
  return all;
}
```
